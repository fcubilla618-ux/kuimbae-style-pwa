// PEGAR DENTRO DE match /databases/{database}/documents
// Conservá las reglas existentes de /users/{userId}.
//
// Acepta únicamente administradores activos para la versión BETA.
// Reutiliza la función isAdmin() ya definida por el proyecto.
function isBetaAdmin() {
  return isAdmin()
    && get(/databases/$(database)/documents/users/$(request.auth.uid))
         .data.get('active', true) == true;
}

match /shops/{shopId}/datasets/{datasetId} {
  allow get, list: if shopId == 'kuimbae-style'
    && isBetaAdmin();

  allow create, update: if shopId == 'kuimbae-style'
    && isBetaAdmin()
    && datasetId in [
      'barbers', 'appointments', 'services', 'cash',
      'clientMeta', 'settings', 'products'
    ]
    && request.resource.data.keys().hasOnly([
      'data', 'version', 'updatedAt', 'updatedBy'
    ])
    && request.resource.data.keys().hasAll([
      'data', 'version', 'updatedAt', 'updatedBy'
    ])
    && request.resource.data.version == 1
    && request.resource.data.updatedBy == request.auth.uid
    && request.resource.data.updatedAt is timestamp
    && (
      (datasetId in ['clientMeta', 'settings']
        && request.resource.data.data is map)
      ||
      (datasetId in ['barbers', 'appointments', 'services',
                     'cash', 'products']
        && request.resource.data.data is list)
    );

  allow delete: if false;
}
